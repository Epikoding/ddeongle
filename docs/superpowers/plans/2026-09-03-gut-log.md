# gut-log Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 말로 받아쓴 한 줄을 사용자의 LLM으로 JSON화해 붙여넣으면 확인 시트를 거쳐 저장하고, 기록 사이의 시간 연관을 "용의자 순위표"로 보여 주는 가족용 PWA.

**Architecture:** Next.js(App Router) 브라우저 앱이 Supabase(Auth·Postgres·RLS)에 supabase-js로 직접 읽고 쓴다. 서버 라우트는 인증 콜백 하나뿐이고 LLM 호출은 없다. 저장(`save_entries`)과 분석(`suspect_ranking`, `suspect_detail`)은 Postgres 함수가 맡는다.

**Tech Stack:** Node 22+, Next.js 16(App Router, `src/`), React 19, TypeScript, Tailwind v4, `@supabase/supabase-js` 2.x, `@supabase/ssr`, `zod` 4.x, `vitest` 3.x, `@playwright/test`, Supabase CLI 2.x(로컬 Postgres, pgTAP).

**Spec:** `/Users/dj/dev/gut-log/docs/superpowers/specs/2026-09-03-gut-log-design.md`

## Global Constraints

- 앱은 LLM·음성인식 API를 호출하지 않는다. 서버 라우트는 `src/app/auth/callback/route.ts` 하나만.
- 모든 사용자 표는 `user_id` 기준 RLS. 함수는 `auth.uid()`로만 사용자를 식별한다.
- 설사는 `bowel.bristol >= 6`으로만 표현한다. `symptom.type`에 "설사"는 없다.
- `entries.kind` 9종: `meal symptom bowel sleep stress exercise medication cycle note`. `time_precision` 3종: `exact approx unknown`.
- 재료 태그는 한글 기본형, `trim()` 후 저장. 감시 재료 목록은 `src/lib/watch-ingredients.ts` 한 곳에만 둔다.
- 분석 창 W ∈ {2,4,8,24}시간, 기본 8. 표적 `diarrhea|urgency|pain|bloating`, 기본 `diarrhea`. 노출 3건 미만은 "자료 부족".
- 기준 시각 표기: `YYYY-MM-DD 요일 HH:MM, 시간대 +09:00` (예 `2026-09-03 목요일 19:45, 시간대 +09:00`). 시간대 기본 `Asia/Seoul`.
- 저장소: `/Users/dj/dev/gut-log`. 커밋 메시지 끝에 `Claude-Session: https://claude.ai/code/session_015j2GFX613gYgCnxsXLLDSF` 한 줄.

```sql
create extension if not exists citext;

create table public.profiles (
  user_id uuid primary key references auth.users(id) on delete cascade,
  display_name text,
  timezone text not null default 'Asia/Seoul',
  created_at timestamptz not null default now()
);

create table public.invites (
  email citext primary key,
  invited_by uuid references auth.users(id) on delete set null,
  created_at timestamptz not null default now()
);

create table public.raw_inputs (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  text text not null,
  source text not null check (source in ('voice','form')),
  reference_at timestamptz not null,
  pasted_json jsonb,
  created_at timestamptz not null default now()
);

create table public.entries (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  raw_input_id uuid references public.raw_inputs(id) on delete set null,
  kind text not null check (kind in ('meal','symptom','bowel','sleep','stress','exercise','medication','cycle','note')),
  occurred_at timestamptz not null,
  time_precision text not null default 'exact' check (time_precision in ('exact','approx','unknown')),
  source_text text,
  data jsonb not null default '{}'::jsonb,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  deleted_at timestamptz
);
create index entries_user_occurred_idx on public.entries (user_id, occurred_at);
create index entries_user_kind_occurred_idx on public.entries (user_id, kind, occurred_at);

create table public.entry_tags (
  entry_id uuid not null references public.entries(id) on delete cascade,
  tag text not null,
  primary key (entry_id, tag)
);

create table public.foods (
  user_id uuid not null references auth.users(id) on delete cascade,
  name text not null,
  ingredients text[] not null default '{}',
  last_used_at timestamptz not null default now(),
  primary key (user_id, name)
);

create or replace function public.handle_new_user() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  insert into public.profiles (user_id, display_name) values (new.id, split_part(new.email, '@', 1));
  return new;
end $$;

create trigger on_auth_user_created after insert on auth.users
  for each row execute function public.handle_new_user();

create or replace function public.touch_updated_at() returns trigger
language plpgsql as $$
begin new.updated_at = now(); return new; end $$;

create trigger entries_touch before update on public.entries
  for each row execute function public.touch_updated_at();
```

create policy invites_insert on public.invites for insert to authenticated with check (invited_by = auth.uid());

create or replace function public.save_entries(p_raw jsonb, p_entries jsonb)
returns uuid[]
language plpgsql security invoker as $$
declare
  v_uid uuid := auth.uid();
  v_raw_id uuid;
  v_ids uuid[] := '{}';
  v_entry jsonb;
  v_id uuid;
  v_food jsonb;
  v_ing text;
  v_ings text[];
begin
  if v_uid is null then
    raise exception 'not_authenticated';
  end if;

  if p_raw is not null then
    insert into public.raw_inputs (user_id, text, source, reference_at, pasted_json)
    values (v_uid, p_raw->>'text', p_raw->>'source', (p_raw->>'reference_at')::timestamptz, p_raw->'pasted_json')
    returning id into v_raw_id;
  end if;

  for v_entry in select * from jsonb_array_elements(p_entries) loop
    insert into public.entries (user_id, raw_input_id, kind, occurred_at, time_precision, source_text, data)
    values (
      v_uid, v_raw_id, v_entry->>'kind', (v_entry->>'occurred_at')::timestamptz,
      coalesce(v_entry->>'time_precision', 'exact'), v_entry->>'source_text', coalesce(v_entry->'data', '{}'::jsonb)
    ) returning id into v_id;
    v_ids := v_ids || v_id;

    if v_entry->>'kind' = 'meal' then
      for v_food in select * from jsonb_array_elements(coalesce(v_entry->'data'->'foods', '[]'::jsonb)) loop
        v_ings := '{}';
        for v_ing in select btrim(x) from jsonb_array_elements_text(coalesce(v_food->'ingredients', '[]'::jsonb)) x loop
          if v_ing <> '' then
            v_ings := array_append(v_ings, v_ing);
            insert into public.entry_tags (entry_id, tag) values (v_id, v_ing) on conflict do nothing;
          end if;
        end loop;
        if coalesce(v_food->>'name', '') <> '' then
          insert into public.foods (user_id, name, ingredients, last_used_at)
          values (v_uid, v_food->>'name', v_ings, now())
          on conflict (user_id, name) do update set ingredients = excluded.ingredients, last_used_at = now();
        end if;
      end loop;
    end if;
  end loop;

  return v_ids;
end $$;

create or replace function public.suspect_ranking(
  p_from timestamptz, p_to timestamptz, p_window_hours int, p_target text
) returns table(
  candidate text, grp text, exposed_n int, exposed_hits int, control_n int, control_hits int, ratio numeric, enough boolean
) language sql stable security invoker as $$
with me as (
  select auth.uid() as uid,
         coalesce((select timezone from public.profiles where user_id = auth.uid()), 'Asia/Seoul') as tz
),
targets as (
  select e.occurred_at
  from public.entries e, me
  where e.user_id = me.uid and e.deleted_at is null
    and e.occurred_at between p_from and p_to
    and case p_target
      when 'diarrhea' then e.kind = 'bowel' and coalesce((e.data->>'bristol')::int, 0) >= 6
      when 'urgency'  then e.kind = 'bowel' and coalesce((e.data->>'urgency')::boolean, false)
      when 'pain'     then e.kind = 'symptom' and e.data->>'type' = '복통'
      when 'bloating' then e.kind = 'symptom' and e.data->>'type' = '팽만'
      else false end
),
meals as (
  select e.id, e.occurred_at, e.data,
    exists (select 1 from targets t
            where t.occurred_at > e.occurred_at
              and t.occurred_at <= e.occurred_at + make_interval(hours => p_window_hours)) as hit
  from public.entries e, me
  where e.user_id = me.uid and e.kind = 'meal' and e.deleted_at is null
    and e.occurred_at between p_from and p_to
),
meal_totals as (select count(*) as n, count(*) filter (where hit) as hits from meals),
meal_cands as (
  select 'ingredient'::text as grp, t.tag as cand, m.id as meal_id, m.hit from meals m join public.entry_tags t on t.entry_id = m.id
  union all
  select 'food', f->>'name', m.id, m.hit from meals m, jsonb_array_elements(coalesce(m.data->'foods','[]'::jsonb)) f where coalesce(f->>'name','') <> ''
  union all
  select 'pattern', p, m.id, m.hit from meals m, jsonb_array_elements_text(coalesce(m.data->'eating_pattern','[]'::jsonb)) p
),
meal_stats as (
  select mc.grp, mc.cand,
    count(distinct mc.meal_id) as exposed_n,
    count(distinct mc.meal_id) filter (where mc.hit) as exposed_hits,
    (select n from meal_totals) - count(distinct mc.meal_id) as control_n,
    (select hits from meal_totals) - count(distinct mc.meal_id) filter (where mc.hit) as control_hits
  from meal_cands mc group by mc.grp, mc.cand
),
days as (
  select d::date as day,
    exists (select 1 from targets t, me where (t.occurred_at at time zone me.tz)::date = d::date) as hit
  from me, generate_series((p_from at time zone me.tz)::date, (p_to at time zone me.tz)::date, interval '1 day') d
),
day_totals as (select count(*) as n, count(*) filter (where hit) as hits from days),
day_cands as (
  select 'sleep'::text as grp, '나쁜 수면'::text as cand, (e.occurred_at at time zone me.tz)::date as day
  from public.entries e, me where e.user_id = me.uid and e.deleted_at is null and e.kind = 'sleep'
    and e.occurred_at between p_from and p_to and coalesce((e.data->>'quality')::int, 10) <= 4
  union all
  select 'stress', '높은 스트레스', (e.occurred_at at time zone me.tz)::date
  from public.entries e, me where e.user_id = me.uid and e.deleted_at is null and e.kind = 'stress'
    and e.occurred_at between p_from and p_to and coalesce((e.data->>'level')::int, 0) >= 6
  union all
  select 'exercise', '고강도 운동', (e.occurred_at at time zone me.tz)::date
  from public.entries e, me where e.user_id = me.uid and e.deleted_at is null and e.kind = 'exercise'
    and e.occurred_at between p_from and p_to and e.data->>'intensity' = '높음'
  union all
  select 'medication', e.data->>'name', (e.occurred_at at time zone me.tz)::date
  from public.entries e, me where e.user_id = me.uid and e.deleted_at is null and e.kind = 'medication'
    and e.occurred_at between p_from and p_to and coalesce(e.data->>'name','') <> ''
),
day_stats as (
  select dc.grp, dc.cand,
    count(distinct dc.day) as exposed_n,
    count(distinct dc.day) filter (where d.hit) as exposed_hits,
    (select n from day_totals) - count(distinct dc.day) as control_n,
    (select hits from day_totals) - count(distinct dc.day) filter (where d.hit) as control_hits
  from day_cands dc join days d on d.day = dc.day group by dc.grp, dc.cand
),
all_stats as (select * from meal_stats union all select * from day_stats)
select cand as candidate, grp,
  exposed_n::int, exposed_hits::int, control_n::int, control_hits::int,
  case when exposed_n > 0 and control_n > 0 and control_hits > 0
       then round((exposed_hits::numeric / exposed_n) / (control_hits::numeric / control_n), 2)
       else null end as ratio,
  (exposed_n >= 3) as enough
from all_stats
order by (exposed_n >= 3) desc, ratio desc nulls last, exposed_n desc, cand;
$$;

create or replace function public.suspect_detail(
  p_from timestamptz, p_to timestamptz, p_window_hours int, p_target text, p_grp text, p_candidate text
) returns table(exposure_id uuid, exposure_at timestamptz, hit boolean, target_id uuid, target_at timestamptz)
language sql stable security invoker as $$
with me as (
  select auth.uid() as uid,
         coalesce((select timezone from public.profiles where user_id = auth.uid()), 'Asia/Seoul') as tz
),
targets as (
  select e.id, e.occurred_at
  from public.entries e, me
  where e.user_id = me.uid and e.deleted_at is null and e.occurred_at between p_from and p_to
    and case p_target
      when 'diarrhea' then e.kind = 'bowel' and coalesce((e.data->>'bristol')::int, 0) >= 6
      when 'urgency'  then e.kind = 'bowel' and coalesce((e.data->>'urgency')::boolean, false)
      when 'pain'     then e.kind = 'symptom' and e.data->>'type' = '복통'
      when 'bloating' then e.kind = 'symptom' and e.data->>'type' = '팽만'
      else false end
),
exposures as (
  select e.id, e.occurred_at
  from public.entries e, me
  where e.user_id = me.uid and e.deleted_at is null and e.occurred_at between p_from and p_to
    and case p_grp
      when 'ingredient' then e.kind = 'meal' and exists (select 1 from public.entry_tags t where t.entry_id = e.id and t.tag = p_candidate)
      when 'food'       then e.kind = 'meal' and exists (select 1 from jsonb_array_elements(coalesce(e.data->'foods','[]'::jsonb)) f where f->>'name' = p_candidate)
      when 'pattern'    then e.kind = 'meal' and e.data->'eating_pattern' ? p_candidate
      when 'sleep'      then e.kind = 'sleep' and coalesce((e.data->>'quality')::int, 10) <= 4
      when 'stress'     then e.kind = 'stress' and coalesce((e.data->>'level')::int, 0) >= 6
      when 'exercise'   then e.kind = 'exercise' and e.data->>'intensity' = '높음'
      when 'medication' then e.kind = 'medication' and e.data->>'name' = p_candidate
      else false end
)
select x.id as exposure_id, x.occurred_at as exposure_at,
  (t.id is not null) as hit, t.id as target_id, t.occurred_at as target_at
from exposures x
left join lateral (
  select t.id, t.occurred_at from targets t, me
  where case when p_grp in ('ingredient','food','pattern')
        then t.occurred_at > x.occurred_at and t.occurred_at <= x.occurred_at + make_interval(hours => p_window_hours)
        else (t.occurred_at at time zone me.tz)::date = (x.occurred_at at time zone me.tz)::date end
  order by t.occurred_at limit 1
) t on true
order by x.occurred_at desc;
$$;

const DAY_KO = ['일요일', '월요일', '화요일', '수요일', '목요일', '금요일', '토요일']

function parts(d: Date, tz: string) {
  const f = new Intl.DateTimeFormat('en-US', {
    timeZone: tz, hourCycle: 'h23',
    year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', weekday: 'short',
    timeZoneName: 'longOffset',
  })
  const p: Record<string, string> = {}
  for (const x of f.formatToParts(d)) p[x.type] = x.value
  const weekday = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'].indexOf(p.weekday)
  const offset = p.timeZoneName === 'GMT' ? '+00:00' : p.timeZoneName.replace('GMT', '')
  return { y: p.year, m: p.month, d: p.day, hh: p.hour, mm: p.minute, weekday, offset }
}

export function formatReference(d: Date, tz: string): string {
  const p = parts(d, tz)
  return `${p.y}-${p.m}-${p.d} ${DAY_KO[p.weekday]} ${p.hh}:${p.mm}, 시간대 ${p.offset}`
}

export function isFuture(iso: string, now: Date, graceMinutes = 5): boolean {
  return new Date(iso).getTime() > now.getTime() + graceMinutes * 60_000
}

export function toLocalDateStr(d: Date, tz: string): string {
  const p = parts(d, tz)
  return `${p.y}-${p.m}-${p.d}`
}

function offsetFor(dateStr: string, tz: string): string {
  return parts(new Date(`${dateStr}T12:00:00Z`), tz).offset
}

export function dayRange(dateStr: string, tz: string): { from: string; to: string } {
  const off = offsetFor(dateStr, tz)
  const from = new Date(`${dateStr}T00:00:00${off}`)
  const to = new Date(from.getTime() + 24 * 3600_000 - 1)
  return { from: from.toISOString(), to: to.toISOString() }
}

import { entrySchema, parsedResultSchema, type Entry } from './schemas'

export type InvalidEntry = { index: number; raw: unknown; reason: string }
export type PasteResult =
  | { ok: true; entries: Entry[]; invalid: InvalidEntry[]; questions: string[] }
  | { ok: false; error: 'no_json' | 'bad_shape'; message: string }

export function extractJson(text: string): string | null {
  const start = text.indexOf('{')
  const end = text.lastIndexOf('}')
  if (start === -1 || end === -1 || end <= start) return null
  return text.slice(start, end + 1)
}

export function parsePaste(text: string): PasteResult {
  const json = extractJson(text)
  if (!json) return { ok: false, error: 'no_json', message: 'JSON을 찾지 못했습니다. LLM 답변 전체를 다시 복사해 주세요.' }

  let obj: unknown
  try {
    obj = JSON.parse(json)
  } catch {
    return { ok: false, error: 'no_json', message: 'JSON이 중간에 끊겼습니다. LLM 답변 전체를 다시 복사해 주세요.' }
  }

  const shape = parsedResultSchema.safeParse(obj)
  if (!shape.success) return { ok: false, error: 'bad_shape', message: '결과 모양이 다릅니다. 프롬프트를 다시 복사해 보내 주세요.' }

  const entries: Entry[] = []
  const invalid: InvalidEntry[] = []
  shape.data.entries.forEach((raw, index) => {
    const r = entrySchema.safeParse(raw)
    if (r.success) entries.push(r.data)
    else invalid.push({ index, raw, reason: r.error.issues.map((i) => `${i.path.join('.')}: ${i.message}`).join('; ') })
  })
  entries.sort((a, b) => a.occurred_at.localeCompare(b.occurred_at))
  return { ok: true, entries, invalid, questions: shape.data.questions }
}

import template from '../../prompts/parse-ko.md'
import { formatReference } from './time'

export type FoodHint = { name: string; ingredients: string[] }

function body(): string {
  const i = template.indexOf('\n---\n')
  return i === -1 ? template : template.slice(i + 5).trimStart()
}

export function buildPrompt(args: { text: string; referenceAt: Date; tz: string; foods?: FoodHint[] }): string {
  let out = body()
    .replace('{{기준_시각}}', formatReference(args.referenceAt, args.tz))
  if (args.foods && args.foods.length > 0) {
    const lines = args.foods.map((f) => `- ${f.name}: ${f.ingredients.join(', ')}`).join('\n')
    const section = `## 이 사용자의 등록 음식\n\n아래 음식은 이 사용자가 재료를 직접 정한 것이다. 이름이 나오면 이 재료 목록을 그대로 쓴다.\n\n${lines}\n\n`
    out = out.replace('## 변환할 문장', `${section}## 변환할 문장`)
  }
  // 마지막 --- 뒤의 "## 변환할 문장" 절에서 두 번째 {{입력_문장}}만 남아 있다
  out = out.replace('{{입력_문장}}', args.text.trim())
  return out
}

import { createServerClient } from '@supabase/ssr'
import { NextResponse, type NextRequest } from 'next/server'

const PUBLIC = ['/login', '/auth/callback', '/manifest.webmanifest', '/sw.js']

export async function updateSession(request: NextRequest) {
  let response = NextResponse.next({ request })
  const supabase = createServerClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!, {
    cookies: {
      getAll() { return request.cookies.getAll() },
      setAll(cookiesToSet) {
        cookiesToSet.forEach(({ name, value }) => request.cookies.set(name, value))
        response = NextResponse.next({ request })
        cookiesToSet.forEach(({ name, value, options }) => response.cookies.set(name, value, options))
      },
    },
  })
  const { data: { user } } = await supabase.auth.getUser()
  const path = request.nextUrl.pathname
  const isPublic = PUBLIC.some((p) => path.startsWith(p)) || path.startsWith('/icons/')
  if (!user && !isPublic) {
    const url = request.nextUrl.clone(); url.pathname = '/login'; return NextResponse.redirect(url)
  }
  if (user && path === '/login') {
    const url = request.nextUrl.clone(); url.pathname = '/'; return NextResponse.redirect(url)
  }
  return response
}

import { NextResponse, type NextRequest } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import type { EmailOtpType } from '@supabase/supabase-js'

export async function GET(request: NextRequest) {
  const url = new URL(request.url)
  const code = url.searchParams.get('code')
  const tokenHash = url.searchParams.get('token_hash')
  const type = url.searchParams.get('type') as EmailOtpType | null
  const supabase = await createClient()

  if (code) {
    const { error } = await supabase.auth.exchangeCodeForSession(code)
    if (!error) return NextResponse.redirect(new URL('/', url.origin))
  } else if (tokenHash && type) {
    const { error } = await supabase.auth.verifyOtp({ token_hash: tokenHash, type })
    if (!error) return NextResponse.redirect(new URL('/', url.origin))
  }
  return NextResponse.redirect(new URL('/login?error=link', url.origin))
}

import type { SupabaseClient } from '@supabase/supabase-js'
import type { Entry } from './schemas'
import type { FoodHint } from './prompt'
import { dayRange } from './time'

export type EntryRow = Entry & { id: string; raw_input_id: string | null; created_at: string }
export type RawInput = { text: string; source: 'voice' | 'form'; reference_at: string; pasted_json: unknown }
export type RankingRow = {
  candidate: string; grp: string; exposed_n: number; exposed_hits: number
  control_n: number; control_hits: number; ratio: number | null; enough: boolean
}
export type DetailRow = { exposure_id: string; exposure_at: string; hit: boolean; target_id: string | null; target_at: string | null }

function must<T>(r: { data: T | null; error: { message: string } | null }): T {
  if (r.error) throw new Error(r.error.message)
  return r.data as T
}

export async function getProfile(sb: SupabaseClient) {
  return must(await sb.from('profiles').select('user_id, display_name, timezone').single()) as
    { user_id: string; display_name: string | null; timezone: string }
}

export async function listEntriesForDay(sb: SupabaseClient, dateStr: string, tz: string): Promise<EntryRow[]> {
  const { from, to } = dayRange(dateStr, tz)
  return must(await sb.from('entries')
    .select('id, raw_input_id, kind, occurred_at, time_precision, source_text, data, created_at')
    .is('deleted_at', null).gte('occurred_at', from).lte('occurred_at', to)
    .order('occurred_at', { ascending: true })) as EntryRow[]
}

export async function getEntry(sb: SupabaseClient, id: string): Promise<EntryRow | null> {
  const r = await sb.from('entries').select('id, raw_input_id, kind, occurred_at, time_precision, source_text, data, created_at')
    .eq('id', id).is('deleted_at', null).maybeSingle()
  if (r.error) throw new Error(r.error.message)
  return r.data as EntryRow | null
}

export async function updateEntry(sb: SupabaseClient, id: string,
  patch: { occurred_at?: string; time_precision?: Entry['time_precision']; data?: Entry['data'] }) {
  must(await sb.from('entries').update(patch).eq('id', id).select('id'))
  if (patch.data && 'foods' in patch.data) {
    const tags = patch.data.foods.flatMap((f) => f.ingredients)
    must(await sb.from('entry_tags').delete().eq('entry_id', id))
    if (tags.length) must(await sb.from('entry_tags').insert([...new Set(tags)].map((tag) => ({ entry_id: id, tag }))))
  }
}

export async function deleteEntry(sb: SupabaseClient, id: string) {
  must(await sb.from('entries').update({ deleted_at: new Date().toISOString() }).eq('id', id).select('id'))
}

export async function getRawInput(sb: SupabaseClient, id: string) {
  const r = await sb.from('raw_inputs').select('text').eq('id', id).maybeSingle()
  if (r.error) throw new Error(r.error.message)
  return r.data as { text: string } | null
}

export async function saveEntries(sb: SupabaseClient, raw: RawInput | null, entries: Entry[]): Promise<string[]> {

}

import type { Entry } from './schemas'
import type { RawInput } from './db'

const KEY = 'gutlog.pending'
export type PendingItem = { id: string; raw: RawInput | null; entries: Entry[]; queued_at: string }

function store(s?: Storage): Storage | null {
  if (s) return s
  try { return typeof window !== 'undefined' ? window.localStorage : null } catch { return null }
}

export function loadQueue(s?: Storage): PendingItem[] {
  const st = store(s); if (!st) return []
  try {
    const v = JSON.parse(st.getItem(KEY) ?? '[]')
    return Array.isArray(v) ? v : []
  } catch { return [] }
}

function write(items: PendingItem[], s?: Storage) {
  const st = store(s); if (!st) return
  try { st.setItem(KEY, JSON.stringify(items)) } catch { /* 저장 공간 부족 등 — 조용히 포기 */ }
}

export function enqueue(item: Omit<PendingItem, 'id' | 'queued_at'>, s?: Storage): PendingItem {
  const full: PendingItem = { ...item, id: crypto.randomUUID(), queued_at: new Date().toISOString() }
  write([...loadQueue(s), full], s)
  return full
}

export function dequeue(id: string, s?: Storage) {
  write(loadQueue(s).filter((i) => i.id !== id), s)
}

  const [items, setItems] = useState<PendingItem[]>([])
  const [busy, setBusy] = useState(false)
  useEffect(() => { setItems(loadQueue()) }, [])

    }

    <div className="mb-3 flex items-center justify-between rounded-lg border border-amber-500 bg-amber-50 p-2 text-sm">

import type { EntryRow } from './db'
import { summarizeEntry } from '@/components/EntryCard'

function q(v: string): string {
  return /[",\n]/.test(v) ? `"${v.replace(/"/g, '""')}"` : v
}

export function entriesToCsv(rows: EntryRow[], tz: string): string {
  const fmt = new Intl.DateTimeFormat('sv-SE', { timeZone: tz, year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', hourCycle: 'h23' })
  const head = 'id,kind,occurred_at_local,time_precision,summary,tags,data_json,source_text'
  const body = rows.map((r) => {
    const tags = r.kind === 'meal' ? [...new Set(r.data.foods.flatMap((f) => f.ingredients))].join('|') : ''
    return [r.id, r.kind, fmt.format(new Date(r.occurred_at)), r.time_precision, summarizeEntry(r), tags, JSON.stringify(r.data), r.source_text ?? ''].map(q).join(',')
  })
  return [head, ...body].join('\n')
}

const CACHE = 'gutlog-shell-v1'
const SHELL = ['/', '/login', '/manifest.webmanifest', '/icons/icon-192.png', '/icons/icon-512.png']

self.addEventListener('install', (e) => {
  e.waitUntil(caches.open(CACHE).then((c) => c.addAll(SHELL)).then(() => self.skipWaiting()))
})
self.addEventListener('activate', (e) => {
  e.waitUntil(caches.keys().then((keys) => Promise.all(keys.filter((k) => k !== CACHE).map((k) => caches.delete(k)))).then(() => self.clients.claim()))
})
self.addEventListener('fetch', (e) => {
  const url = new URL(e.request.url)
  if (e.request.method !== 'GET' || url.origin !== self.location.origin) return
  if (url.pathname.startsWith('/_next/static/') || url.pathname.startsWith('/icons/')) {
    e.respondWith(caches.open(CACHE).then(async (c) => (await c.match(e.request)) || fetch(e.request).then((r) => { c.put(e.request, r.clone()); return r })))
    return
  }
  e.respondWith(fetch(e.request).catch(() => caches.match(e.request).then((r) => r || caches.match('/'))))
})

import { createClient } from '@supabase/supabase-js'
import { execSync } from 'node:child_process'

function localKeys() {
  const out = execSync('supabase status -o json', { cwd: process.cwd() }).toString()
  const j = JSON.parse(out)
  return { url: j.API_URL as string, service: j.SERVICE_ROLE_KEY as string }
}

export async function loginUrlFor(email: string): Promise<string> {
  const { url, service } = localKeys()
  const admin = createClient(url, service, { auth: { persistSession: false } })
  await admin.from('invites').upsert({ email })
  const { data, error } = await admin.auth.admin.generateLink({ type: 'magiclink', email })
  if (error) throw error
  return `/auth/callback?token_hash=${data.properties.hashed_token}&type=magiclink`
}
